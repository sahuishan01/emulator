cmake_minimum_required(VERSION 3.22)
project(emulator_gpu_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile=2 link=1)
set(CMAKE_JOB_POOL_COMPILE compile)
set(CMAKE_JOB_POOL_LINK link)

if(ANDROID)
    add_library(emulator_gpu_lab SHARED src/gpu_lab.cpp src/jni.cpp)
    target_link_libraries(emulator_gpu_lab PRIVATE vulkan)
    target_link_options(emulator_gpu_lab PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(Vulkan REQUIRED)
    add_library(emulator_gpu_lab STATIC src/gpu_lab.cpp)
    target_link_libraries(emulator_gpu_lab PUBLIC Vulkan::Vulkan)
    add_executable(gpu-lab-cli src/main.cpp)
    target_link_libraries(gpu-lab-cli PRIVATE emulator_gpu_lab)
endif()
target_include_directories(emulator_gpu_lab PRIVATE generated)
target_compile_options(emulator_gpu_lab PRIVATE -Wall -Wextra -Werror)
