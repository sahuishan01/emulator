cmake_minimum_required(VERSION 3.22)
project(emulator_gpu_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile=2 link=1)
set(CMAKE_JOB_POOL_COMPILE compile)
set(CMAKE_JOB_POOL_LINK link)

if(ANDROID)
    if(NOT EMULATOR_ENGINE_LIB_DIR)
        set(EMULATOR_ENGINE_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/android-engine")
    endif()
    set(ENGINE_ARCHIVE "${EMULATOR_ENGINE_LIB_DIR}/${ANDROID_ABI}/libemulator_android_ffi.a")
    if(NOT EXISTS "${ENGINE_ARCHIVE}")
        message(FATAL_ERROR "Original Rust engine is missing. Run tools/build_android_engine.py first.")
    endif()
    add_library(emulator_original_engine STATIC IMPORTED)
    set_target_properties(emulator_original_engine PROPERTIES IMPORTED_LOCATION "${ENGINE_ARCHIVE}")
    add_library(emulator_gpu_lab SHARED src/gpu_lab.cpp src/jni.cpp)
    target_link_libraries(emulator_gpu_lab PRIVATE emulator_original_engine vulkan log dl m)
    target_link_options(emulator_gpu_lab PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(Vulkan REQUIRED)
    add_library(emulator_gpu_lab STATIC src/gpu_lab.cpp)
    target_link_libraries(emulator_gpu_lab PUBLIC Vulkan::Vulkan)
    add_executable(gpu-lab-cli src/main.cpp)
    target_link_libraries(gpu-lab-cli PRIVATE emulator_gpu_lab)
endif()
target_include_directories(emulator_gpu_lab PRIVATE generated)
target_compile_options(emulator_gpu_lab PRIVATE -Wall -Wextra -Werror)
